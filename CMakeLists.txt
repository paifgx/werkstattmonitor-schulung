cmake_minimum_required(VERSION 3.21)
project(Werkstattmonitor VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
option(LAB_CONTRACT_TESTS "Zielverhalten aktivieren (im Ausgangsstand absichtlich rot)" OFF)
find_package(Qt6 6.2 REQUIRED COMPONENTS Core Widgets Test)
include(CTest)
add_library(telemetry src/legacy.cpp src/telemetry_file.cpp src/telemetry_summary.cpp src/telemetry.h)
target_include_directories(telemetry PUBLIC src)
target_link_libraries(telemetry PUBLIC Qt6::Core)
add_library(hmi src/hmiwindow.cpp src/hmiwindow.h)
target_link_libraries(hmi PUBLIC telemetry Qt6::Widgets)
add_executable(werkstattmonitor src/main.cpp)
target_link_libraries(werkstattmonitor PRIVATE hmi)
add_custom_command(TARGET werkstattmonitor POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory
    "${CMAKE_CURRENT_SOURCE_DIR}/data" "$<TARGET_FILE_DIR:werkstattmonitor>/data")
foreach(target telemetry hmi werkstattmonitor)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
if(BUILD_TESTING)
    add_executable(characterization tests/characterization.cpp)
    target_link_libraries(characterization PRIVATE telemetry Qt6::Test)
    add_test(NAME baseline.parser COMMAND characterization)
    add_executable(widget_tests tests/widget_tests.cpp)
    target_link_libraries(widget_tests PRIVATE hmi Qt6::Test)
    if(MSVC)
        target_compile_options(characterization PRIVATE /utf-8)
        target_compile_options(widget_tests PRIVATE /utf-8)
    endif()
    add_test(NAME baseline.widget COMMAND widget_tests smoke)
    set_tests_properties(baseline.widget PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    if(LAB_CONTRACT_TESTS)
        add_executable(parser_contract tests/parser_contract.cpp)
        target_link_libraries(parser_contract PRIVATE telemetry Qt6::Test)
        if(MSVC)
            target_compile_options(parser_contract PRIVATE /utf-8)
        endif()
        add_test(NAME contract.parser COMMAND parser_contract)
        add_test(NAME contract.widget COMMAND widget_tests)
        set_tests_properties(contract.widget PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    endif()
    get_property(labTests DIRECTORY PROPERTY TESTS)
    set_tests_properties(${labTests} PROPERTIES TIMEOUT 20)
endif()
